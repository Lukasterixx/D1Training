#!/usr/bin/env python3
"""Send funcode 6 power=0 repeatedly at 10 Hz and watch whether power_status drops."""
import json, os, time
from dataclasses import dataclass
os.environ["CYCLONEDDS_URI"] = ("<CycloneDDS><Domain><General><Interfaces>"
                                "<NetworkInterface name='enP8p1s0'/>"
                                "</Interfaces></General></Domain></CycloneDDS>")
from cyclonedds.core import ReadCondition, InstanceState, SampleState, ViewState, WaitSet
from cyclonedds.domain import DomainParticipant
from cyclonedds.idl import IdlStruct
from cyclonedds.pub import DataWriter
from cyclonedds.sub import DataReader
from cyclonedds.topic import Topic
from cyclonedds.util import duration


@dataclass
class ArmString(IdlStruct, typename="unitree_arm::msg::dds_::ArmString_"):
    data_: str = ""


dp = DomainParticipant(0)
w = DataWriter(dp, Topic(dp, "rt/arm_Command", ArmString))
r = DataReader(dp, Topic(dp, "rt/arm_Feedback", ArmString))
ws = WaitSet(dp)
ws.attach(ReadCondition(r, SampleState.Any | ViewState.Any | InstanceState.Any))

t0 = time.monotonic()
seq = 0
last = None
n_sent = 0
while time.monotonic() - t0 < 12.0:
    now = time.monotonic() - t0
    if 2.0 < now < 7.0:
        seq = (seq % 9999) + 1
        w.write(ArmString(data_=json.dumps(
            {"seq": seq, "address": 1, "funcode": 6, "data": {"power": 0}})))
        n_sent += 1
    try:
        ws.wait(duration(milliseconds=100))
    except Exception:
        pass
    for s in r.take(N=100):
        try:
            m = json.loads(s.data_)
        except Exception:
            continue
        d = m.get("data") or {}
        if m.get("funcode") == 3:
            cur = (d.get("enable_status"), d.get("power_status"), d.get("error_status"))
            if cur != last:
                print(f"[{now:6.3f}] enable={cur[0]} power={cur[1]} error={cur[2]}")
                last = cur
print(f"sent {n_sent} power=0 commands; final enable={last[0]} power={last[1]} error={last[2]}")
