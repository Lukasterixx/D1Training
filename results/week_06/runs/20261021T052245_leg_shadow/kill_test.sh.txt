#!/usr/bin/env bash
# $1: signal (TERM or KILL); $2: seconds before it; $3: run dir.
env -i HOME="$HOME" PATH=/usr/bin:/bin bash -c '
source /opt/ros/humble/setup.bash && source $HOME/unitree_ros2/install/setup.bash
export ROS_LOCALHOST_ONLY=1 ROS_DOMAIN_ID=77 PYTHONUNBUFFERED=1
python3 '"$(dirname "$0")"'/fake_lowstate.py 30 >/dev/null 2>&1 & FAKE=$!
sleep 2
cd /home/lukas/D1Training
python3 demos/combiner_key/robot/shadow.py --policy checkpoints/leg_key_force_s2walker_model_2104.npz --arm_link_port 0 \
  --print_s 100 --run_dir '"$3"' > '"$3"'.log 2>&1 & NODE=$!
sleep '"$2"'; kill -'"$1"' $NODE; wait $NODE; echo "exit $?"
kill $FAKE 2>/dev/null; wait $FAKE 2>/dev/null
tail -2 '"$3"'.log
'
